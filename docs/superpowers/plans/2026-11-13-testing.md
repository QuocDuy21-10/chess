# Rule, Sequencer and Cross-Browser E2E Tests Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** One shared catalog of chess games and refused moves, replayed by the shared rules, the server, the site's store and the animation sequencer; and a Playwright suite that plays the real game in Chromium, Firefox and WebKit on every pull request, gating deploys.

**Architecture:** `packages/shared/src/testing/games.ts` exports `GAMES` and `REFUSALS`; each layer's existing Vitest file gains a loop over them. A new workspace `e2e/` runs `@playwright/test` against the production site (`next start`) and the real game server, signed in through sessions that `e2e/seed.ts` writes into Postgres. CI gets a matrix job `e2e` that `image` waits for.

**Tech Stack:** Vitest 5, chess.js 1.4, `@playwright/test` 1.64.1, Next.js 16.4, GitHub Actions.

**Spec:** `docs/superpowers/specs/2026-11-13-testing-design.md`

```ts
import type { Chess } from 'chess.js';
import type { Result } from '../result.ts';

// Named games and refused moves that every layer replays: the shared rules, the server, the store and the
// sequencer. For tests only. Moves are in coordinate form from the starting position.

export interface CatalogGame {
  name: string;
  moves: string[];
  result: Result | null; // how the game stands after the last move; null while it goes on
}

export interface Refusal {
  name: string;
  moves: string[]; // all legal, played first
  refused: string; // then this one, which must be refused
}

const list = (moves: string) => moves.split(' ');

export const GAMES: CatalogGame[] = [
  { name: 'fools-mate', moves: list('f2f3 e7e5 g2g4 d8h4'), result: { reason: 'checkmate', winner: 'b' } },
  {
    name: 'scholars-mate',
    moves: list('e2e4 e7e5 f1c4 b8c6 d1h5 g8f6 h5f7'),
    result: { reason: 'checkmate', winner: 'w' },
  },
  { name: 'castles', moves: list('e2e4 d7d5 g1f3 b8c6 f1e2 c8f5 e1g1 d8d6 d2d3 e8c8'), result: null },
  { name: 'en-passant', moves: list('e2e4 a7a6 e4e5 d7d5 e5d6 h7h5 a2a3 h5h4 g2g4 h4g3'), result: null },
  { name: 'underpromotion', moves: list('h2h4 g7g5 h4g5 h7h6 g5h6 e7e6 h6h7 d8e7 h7g8n'), result: null },
  { name: 'promotion', moves: list('h2h4 g7g5 h4g5 g8f6 g5g6 e7e6 g6g7 d8e7 g7g8q'), result: null },
  {
    name: 'stalemate',
    moves: list('e2e3 a7a5 d1h5 a8a6 h5a5 h7h5 h2h4 a6h6 a5c7 f7f6 c7d7 e8f7 d7b7 d8d3 b7b8 d3h7 b8c8 f7g6 c8e6'),
    result: { reason: 'stalemate' },
  },
  { name: 'threefold', moves: list('g1f3 g8f6 f3g1 f6g8 g1f3 g8f6 f3g1 f6g8'), result: { reason: 'threefold' } },
];

export const REFUSALS: Refusal[] = [
  { name: 'castling out of check', moves: list('e2e4 d7d5 e4d5 d8d5 g1f3 d5e5 f1e2 e5e2'), refused: 'e1g1' },
  { name: 'castling through an attacked square', moves: list('e2e4 b7b6 g2g3 c8a6 f1g2 e7e6 g1f3 d7d6'), refused: 'e1g1' },
  { name: 'castling after the king has moved', moves: list('e2e4 e7e5 g1f3 g8f6 f1e2 f8e7 e1f1 e8f8 f1e1 f8e8'), refused: 'e1g1' },
  { name: 'en passant one move late', moves: list('e2e4 a7a6 e4e5 d7d5 a2a3 a6a5'), refused: 'e5d6' },

/** Plays coordinate-form moves on `chess`. Throws on an illegal one, as chess.js does. */
export function playMoves(chess: Chess, moves: string[]): void {
  for (const move of moves) chess.move({ from: move.slice(0, 2), to: move.slice(2, 4), promotion: move[4] });
}
```

- [ ] **Step 3: Write the failing tests for `gameResult`**

Create `packages/shared/src/result.test.ts`:
```ts
import { Chess } from 'chess.js';
import { expect, test } from 'vitest';
import { gameResult } from './result.ts';
import { GAMES, playMoves, REFUSALS } from './testing/games.ts';

test.each(GAMES)('$name ends as the catalog says, and not before its last move', ({ moves, result }) => {
  const chess = new Chess();
  playMoves(chess, moves.slice(0, -1));
  expect(gameResult(chess)).toBeNull();
  playMoves(chess, moves.slice(-1));
  expect(gameResult(chess)).toEqual(result);
});

test.each(REFUSALS)('$name: chess.js refuses the move after a legal start', ({ moves, refused }) => {
  const chess = new Chess();
  playMoves(chess, moves);
  expect(() => playMoves(chess, [refused])).toThrow();
});

/** The result once `move` (coordinate form, optional) is played on `fen`. */
function after(fen: string, move?: string) {
  const chess = new Chess(fen);
  if (move) playMoves(chess, [move]);
  return gameResult(chess);
}

test('the fifty-move rule draws on the hundredth half-move without a capture or a pawn move', () => {
  expect(after('8/8/8/8/8/5k2/8/R3K3 w - - 98 60', 'a1a2')).toBeNull();
  expect(after('8/8/8/8/8/5k2/8/R3K3 w - - 99 60', 'a1a2')).toEqual({ reason: 'fifty' });
});

test('a mate on the hundredth half-move is a checkmate, not a draw', () => {
  expect(after('6k1/5ppp/8/8/8/8/8/R5K1 w - - 99 60', 'a1a8')).toEqual({ reason: 'checkmate', winner: 'w' });
});

test('insufficient material: a lone king, a knight or a bishop, or bishops on one colour', () => {
  expect(after('8/8/8/4k3/8/8/8/4K3 w - - 0 1')).toEqual({ reason: 'insufficient' });
  expect(after('8/8/8/4k3/8/8/8/3NK3 w - - 0 1')).toEqual({ reason: 'insufficient' });
  expect(after('8/8/8/4k3/8/8/8/3BK3 w - - 0 1')).toEqual({ reason: 'insufficient' });
  expect(after('4bk2/8/8/8/8/8/8/3BK3 w - - 0 1')).toEqual({ reason: 'insufficient' });
  expect(after('3b1k2/8/8/8/8/8/8/3BK3 w - - 0 1')).toBeNull(); // bishops on different colours can still mate
});

test('a check alone does not end the game', () => {
  const chess = new Chess();
  playMoves(chess, ['e2e4', 'f7f6', 'd1h5']);
  expect(chess.inCheck()).toBe(true);
  expect(gameResult(chess)).toBeNull();
});
```

```ts
/** Plays `moves` from the start through planMove and runSteps; returns the log of the last move and the pieces. */
async function replay(moves: string[]) {
  const chess = new Chess();
  const { log, deps, pieces } = setup(DEFAULT_POSITION);
  let last: string[] = [];
  for (const text of moves) {
    const move = chess.move({ from: text.slice(0, 2), to: text.slice(2, 4), promotion: text[4] });
    const start = log.length;
    await runSteps(planMove(move, pieces()), deps);
    last = log.slice(start);
    expect(matchesBoard(pieces(), chess), text).toBe(true);
  }
  return { last, pieces: pieces() };
}

test.each(GAMES)('$name animated move by move leaves the pieces where chess.js has them', async ({ moves }) => {
  const { pieces } = await replay(moves);
  const ids = pieces.map((p) => p.id);
  expect(new Set(ids).size).toBe(ids.length); // no piece lost or duplicated its id
  const board = new Chess();
  playMoves(board, moves);
  expect(matchesBoard(pieces, board)).toBe(true);
});

test('a castle walks the king before the rook hops', async () => {
  const { last } = await replay(GAMES.find((g) => g.name === 'castles')!.moves.slice(0, 7)); // up to e1g1
  expect(last.indexOf('w-k-e1 walk end')).toBeLessThan(last.indexOf('w-r-h1 hop start'));
});

test('en passant fights the pawn beside the target square, then steps in', async () => {
  const { last, pieces } = await replay(GAMES.find((g) => g.name === 'en-passant')!.moves.slice(0, 5)); // up to e5d6
  expect(last.indexOf('b-p-d7 die start')).toBeGreaterThan(-1);
  expect(last.lastIndexOf('w-p-e2 walk start')).toBeGreaterThan(last.indexOf('b-p-d7 die end'));
  expect(pieces.find((p) => p.id === 'w-p-e2')?.square).toBe('d6');
});

test('a capture that promotes fights, walks, then promotes, and the piece keeps its id', async () => {
  const { last, pieces } = await replay(GAMES.find((g) => g.name === 'underpromotion')!.moves);
  expect(last.indexOf('w-p-h2 attack start')).toBeGreaterThan(-1);
  expect(last.lastIndexOf('w-p-h2 walk end')).toBeGreaterThan(last.indexOf('b-n-g8 die end'));
  expect(pieces.find((p) => p.id === 'w-p-h2')).toMatchObject({ type: 'n', square: 'g8' });
});
```

- [ ] **Step 2: The seed**

Create `e2e/seed.ts`:
```ts
import { mkdirSync, writeFileSync } from 'node:fs';
import { connectTest, truncate } from '@chess/server/test-db';
import { createSessions } from '../apps/server/src/sessions.ts';

// Run by Node before Playwright: an empty test database, two players with sessions, and a storage state each,
// so the browsers are signed in without OAuth.

const { pool, db } = connectTest(); // throws, naming TEST_DATABASE_URL, when it is not set
try {
  await truncate(pool);
  const sessions = createSessions(db);
  mkdirSync('.auth', { recursive: true });
  for (const [file, name] of [['ann', 'E2E Ann'], ['bob', 'E2E Bob']]) {
    const { rows } = await pool.query('INSERT INTO users (display_name) VALUES ($1) RETURNING id', [name]);
    const token = await sessions.create(rows[0].id);
    const cookie = { name: 'sid', value: token, domain: 'localhost', path: '/', expires: -1, httpOnly: true, secure: false, sameSite: 'Lax' };
    writeFileSync(`.auth/${file}.json`, JSON.stringify({ cookies: [cookie], origins: [] }));
  }
} finally {
  await pool.end();
}
```
Note: the game server applies migrations when it starts, but the seed runs first; `chess_test` is migrated by the Vitest global setup already. If the seed fails with "relation users does not exist" on a fresh database, run `npm test -w @chess/server` once (it migrates), or import `migrate` from `../apps/server/src/db.ts` and call `await migrate(pool)` before `truncate` — do the latter, since CI starts from an empty database.

So the final `seed.ts` body starts with:

Create `e2e/fixtures.ts`:
```ts
import { test as base, expect, type Page } from '@playwright/test';

// Every page a test opens is watched: a page error, a console error or a CSP violation fails the test, and the
// 3D scene must get a WebGL context.

/** Starts recording problems on `page`; returns the list they are added to. */
export function watch(page: Page): string[] {
  const problems: string[] = [];
  page.on('pageerror', (error) => problems.push(`page error: ${error.message}`));
  page.on('console', (message) => {
    if (message.type() === 'error') problems.push(`console error: ${message.text()}`);
    if (message.text().startsWith('CSP ')) problems.push(message.text());
  });
  return problems;
}

/** Waits for the scene's canvas and checks it has a WebGL context. */
export async function expectWebGL(page: Page) {
  await page.locator('canvas').first().waitFor();
  const webgl = await page.evaluate(() => !!document.querySelector('canvas')?.getContext('webgl2'));
  expect(webgl, 'the canvas has a WebGL2 context').toBe(true);
}

export const test = base.extend<{ problems: string[] }>({
  context: async ({ context }, use) => {
    await context.addInitScript(() =>
      document.addEventListener('securitypolicyviolation', (event) =>
        console.log(`CSP ${event.effectiveDirective} ${event.blockedURI}`),
      ),
    );
    await use(context);
  },
  problems: [
    async ({ page }, use) => {
      const problems = watch(page);
      await use(problems);
      expect(problems, 'problems on the page').toEqual([]);
    },
    { auto: true },
  ],
});

export { expect };

/** Types `move` into the HUD and presses Move. */
export async function typeMove(page: Page, move: string) {
  await page.getByLabel('Type a move').fill(move);
  await page.getByRole('button', { name: 'Move', exact: true }).click();
}

/** The number of moves in the HUD's move list. */
export const moveCount = (page: Page) => page.getByRole('list', { name: 'Moves' }).first().getByRole('listitem').count();
```

- [ ] **Step 1: Add the job**

```yaml
  e2e:
    if: github.event_name != 'workflow_dispatch' # a rollback ships an image that already passed
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        browser: [chromium, firefox, webkit]
    name: e2e (${{ matrix.browser }})
    services:
      postgres:
        image: postgres:17-alpine
        env:
          POSTGRES_USER: chess
          POSTGRES_PASSWORD: chess
          POSTGRES_DB: chess_test
        ports: ['5432:5432']
        options: >-
          --health-cmd "pg_isready -U chess"
          --health-interval 5s
          --health-timeout 5s
          --health-retries 10
    env:
      TEST_DATABASE_URL: postgres://chess:chess@localhost:5432/chess_test
    steps:
      - uses: actions/checkout@v7
      - uses: actions/setup-node@v7
        with:
          node-version: 22
          cache: npm
      - run: npm ci
      - run: npm run build -w @chess/web
        env:
          NEXT_PUBLIC_SERVER_URL: http://localhost:3001
      - uses: actions/cache@v4

            e2e/playwright-report
```
